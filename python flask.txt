def _parse_options_json(options_json):
    if not options_json:
        return []
    try:
        obj = json.loads(options_json)
        if isinstance(obj, dict):
            return list(obj.values())
        if isinstance(obj, list):
            return obj
    except Exception:
        pass
    return []


def get_student_id_by_arid(arid_no: str):

def get_profile_value(student_id: int, feature_key: str):
    row = db.session.execute(
        text("""
                SELECT TOP 1 spr.Response_Value
                FROM dbo.Student_Profile_Response spr
                JOIN dbo.Profile_Questions pq
                    ON pq.Profile_Question_ID = spr.Profile_Question_ID
                WHERE spr.Student_ID = :sid
                  AND pq.Feature_Key = :fk
                ORDER BY spr.Created_On DESC
            """),
        {"sid": student_id, "fk": feature_key},
    ).fetchone()
    return row[0] if row else None

def calculate_full_weighted_scores(test_id: int):
    # --- 1. Pull Gate answers with weights ---
    gate_rows = db.session.execute(text("""
        SELECT c.Category_Key, c.Category_Name, COALESCE(c.Category_Weight, 1.0) AS cat_weight,
               COALESCE(cg.Question_Weight, 1.0) AS q_weight, CAST(a.Answer_Value AS FLOAT) AS score
        FROM dbo.Student_Test_Answer a
        JOIN dbo.Category_Gates cg ON cg.Gate_ID = a.Ref_ID
        JOIN dbo.Categories c ON c.Category_ID = cg.Category_ID
        WHERE a.Test_ID = :tid AND a.Item_Type = 'Gate'
    """), {"tid": int(test_id)}).fetchall()

    # --- 2. Pull Base deep-dive answers with weights ---
    base_rows = db.session.execute(text("""
        SELECT c.Category_Key, c.Category_Name, COALESCE(c.Category_Weight, 1.0) AS cat_weight,
               COALESCE(b.Weight, 1.0) AS q_weight, CAST(a.Answer_Value AS FLOAT) AS score
        FROM dbo.Student_Test_Answer a
        JOIN dbo.DeepDive_Base_Questions b ON b.Base_ID = a.Ref_ID
        JOIN dbo.Categories c ON c.Category_ID = b.Category_ID
        WHERE a.Test_ID = :tid AND a.Item_Type = 'Base'
    """), {"tid": int(test_id)}).fetchall()

    # --- 3. Pull Sub deep-dive answers with weights ---
    sub_rows = db.session.execute(text("""
        SELECT c.Category_Key, c.Category_Name, COALESCE(c.Category_Weight, 1.0) AS cat_weight,
               COALESCE(s.Weight, 1.0) AS q_weight, CAST(a.Answer_Value AS FLOAT) AS score
        FROM dbo.Student_Test_Answer a
        JOIN dbo.DeepDive_Sub_Questions s ON s.Sub_ID = a.Ref_ID
        JOIN dbo.DeepDive_Base_Questions b ON b.Base_ID = s.Base_ID
        JOIN dbo.Categories c ON c.Category_ID = b.Category_ID
        WHERE a.Test_ID = :tid AND a.Item_Type = 'Sub'
    """), {"tid": int(test_id)}).fetchall()

    # --- 4. Aggregate per category ---
    cat_data = {}

    def _add_row(ck, cn, cat_w, q_w, score):
        if ck not in cat_data:
            cat_data[ck] = {"Category_Key": ck, "Category_Name": cn, "cat_weight": float(cat_w),
                            "raw_sum": 0.0, "max_possible": 0.0, "question_count": 0}
        entry = cat_data[ck]
        entry["raw_sum"] += float(score) * float(q_w)
        entry["max_possible"] += 4.0 * float(q_w)
        entry["question_count"] += 1

    for rows in [gate_rows, base_rows, sub_rows]:
        for ck, cn, cw, qw, sc in rows:
            _add_row(ck, cn, cw, qw, sc)

    for entry in cat_data.values():
        entry["normalized_pct"] = round((entry["raw_sum"] / entry["max_possible"]) * 100, 2) if entry[
                                                                                                    "max_possible"] > 0 else 0.0

    # --- 5. Compute bucket-level weighted contributions ---
    buckets = {"stress": {"weighted_sum": 0.0, "weight_total": 0.0},
               "anxiety": {"weighted_sum": 0.0, "weight_total": 0.0},
               "depression": {"weighted_sum": 0.0, "weight_total": 0.0}}

    for entry in cat_data.values():
        bucket = CATEGORY_TO_BUCKET.get(entry["Category_Key"])
        if bucket:
            cw = entry["cat_weight"]
            buckets[bucket]["weighted_sum"] += entry["normalized_pct"] * cw
            buckets[bucket]["weight_total"] += cw

    bucket_pct = {bname: round(bdata["weighted_sum"] / bdata["weight_total"], 2) if bdata["weight_total"] > 0 else 0.0
                  for bname, bdata in buckets.items()}

    # --- 6. SH Special Logic: Collect the MAX score (not average) ---
    sh_gate_score = next((float(sc) for ck, cn, cw, qw, sc in gate_rows if ck == "SH"), None)
    sh_base_max = max([float(sc) for ck, cn, cw, qw, sc in base_rows if ck == "SH"], default=None)
    sh_sub_max = max([float(sc) for ck, cn, cw, qw, sc in sub_rows if ck == "SH"], default=None)

    return {
        "bucket_pct": bucket_pct,
        "cat_data": cat_data,
        "sh_gate_score": sh_gate_score,
        "sh_base_max": sh_base_max,
        "sh_sub_max": sh_sub_max
    }


def determine_severity(bucket_pct: dict, sh_gate_score, sh_base_max, sh_sub_max):
    max_pct = max(bucket_pct.values()) if bucket_pct else 0
    dominant = max(bucket_pct, key=bucket_pct.get) if bucket_pct else "stress"

    if max_pct < 20:
        severity = "general_wellness"
    elif max_pct < 40:
        severity = "mild"
    elif max_pct < 60:
        severity = "moderate"
    elif max_pct < 80:
        severity = "high"
    else:
        severity = "severe"

    # --- STRICT SH OVERRIDE ---
    # If ANY SH question (gate, base, or sub) is >= 1 (Rarely), instantly lock to "severe"
    if (sh_gate_score is not None and sh_gate_score >= 1) or \
            (sh_base_max is not None and sh_base_max >= 1) or \
            (sh_sub_max is not None and sh_sub_max >= 1):
        severity = "severe"
        dominant = "anxiety"  # SH always maps to anxiety

    return dominant.capitalize(), severity


def build_test_summary(test_id: int):
    scored = calculate_full_weighted_scores(int(test_id))
    bucket_pct = scored["bucket_pct"]

    dominant_disease, severity = determine_severity(
        bucket_pct, scored["sh_gate_score"], scored["sh_base_max"], scored["sh_sub_max"]
    )

    if severity == "general_wellness":
        pie_pct = {"stress": 0, "anxiety": 0, "depression": 0}
    else:
        active = {k: v for k, v in bucket_pct.items() if v > 0}
        total = sum(active.values())
        normalized = {k: round((v / total) * 100) for k, v in active.items()} if total > 0 else {"stress": 0,
                                                                                                 "anxiety": 0,
                                                                                                 "depression": 0}
        for k in ["stress", "anxiety", "depression"]: normalized.setdefault(k, 0)

        current = sum(normalized.values())
        if current != 100 and current > 0:
            max_bucket = max(normalized, key=normalized.get)
            normalized[max_bucket] += (100 - current)
        pie_pct = normalized

    triggered_all = get_triggered_categories_for_test(int(test_id))
    triggered_sorted = sorted(triggered_all, key=lambda x: x["Gate_Score"], reverse=True)
    sh_item = next((x for x in triggered_sorted if x["Category_Key"] == "SH"), None)
    non_sh = [x for x in triggered_sorted if x["Category_Key"] != "SH"]

    triggered_categories = ([sh_item] + non_sh[:2]) if sh_item else triggered_sorted[:3]

    return {
        "triggered_all": triggered_all,
        "triggered_categories": triggered_categories,
        "triggered_count": len(triggered_all),
        "pie": pie_pct,
        "Dominant_Disease": dominant_disease,
        "Severity": severity,
        "bucket_pct": bucket_pct,
        "cat_data": scored["cat_data"]
    }


def get_triggered_categories_for_test(test_id: int):
    rows = db.session.execute(
        text("""
                SELECT c.Category_Key, c.Category_Name, a.Answer_Value
                FROM dbo.Student_Test_Answer a
                JOIN dbo.Category_Gates g ON g.Gate_ID = a.Ref_ID
                JOIN dbo.Categories c ON c.Category_ID = g.Category_ID
                WHERE a.Test_ID = :tid
                  AND a.Item_Type = 'Gate'
            """),
        {"tid": int(test_id)},
    ).fetchall()

    triggered_all = []
    for ck, cn, val in rows:
        val = int(val)
        cn = cn or ""
        short_name = cn.split(" (")[0].strip()

        if ck == "SH":
            if val >= 1:
                triggered_all.append({
                    "Category_Key": ck,
                    "Category_Name": cn,
                    "Category_Short_Name": short_name,
                    "Gate_Score": val,
                })
        else:
            if val >= 3:
                triggered_all.append({
                    "Category_Key": ck,
                    "Category_Name": cn,
                    "Category_Short_Name": short_name,
                    "Gate_Score": val,
                })

    return triggered_all


def map_profile_context(student_id: int):
    gender = (get_profile_value(student_id, "gender") or "Unknown").strip()
    live_with_family = (get_profile_value(student_id, "live_with_family") or "Unknown").strip()
    family_income_bucket = (get_profile_value(student_id, "family_income_bucket") or "Unknown").strip()

    living_situation = "With Family" if live_with_family.lower() == "yes" else (
        "Away From Family" if live_with_family.lower() == "no" else "Unknown"
    )

    if family_income_bucket in ["0-50k", "50k-100k"]:
        income_context = "Financial Pressure Possible"
    elif family_income_bucket in ["100k-200k", "200k+"]:
        income_context = "Stable Income Context"
    else:
        income_context = "Income Unknown"

    return {
        "gender": gender,
        "living_situation": living_situation,
        "income_context": income_context,
        "live_with_family_raw": live_with_family,
        "family_income_bucket_raw": family_income_bucket,
    }


def is_emergency_case(summary):
    triggered_all = summary["triggered_all"]
    dominant = (summary["Dominant_Disease"] or "").lower()
    severity = (summary["Severity"] or "").lower()

    if any(x["Category_Key"] == "SH" for x in triggered_all):
        return True

    if severity == "high" and dominant in ["depression", "anxiety"]:
        return True

    return False

def safe_fetch_remedies(summary, student_context, max_duration=30):
    """
    Fetches actual remedies directly from the SQL database.
    Limits to exactly 4 remedies per request to prevent overwhelm,
    and randomizes them so the student sees fresh options each day.
    """
    dominant = summary["Dominant_Disease"]
    severity = summary["Severity"].lower()

    try:
        # Added 'TOP 4' to strictly limit the amount shown
        sql = text("""
                SELECT TOP 4
                    Remedy_ID,
                    Title,
                    Description,
                    Remedy_Type,
                    Duration_Minutes,
                    Target_Severity,
                    Resource_Link,
                    Linked_Sub_Category_ID
                FROM dbo.Remedy_Library
                WHERE Is_Active = 1
                  AND (
                        Target_Disease IS NULL
                        OR Target_Disease = :dom
                        OR Target_Disease = 'General Wellness'
                      )
                  AND (
                        Target_Severity IS NULL
                        OR LOWER(Target_Severity) = :sev
                        OR LOWER(Target_Severity) = 'all'
                      )
                  AND (
                        Duration_Minutes IS NULL
                        OR Duration_Minutes <= :maxd
                        OR :maxd >= 999
                      )
                ORDER BY
                    -- Priority 1: Match their exact disease first
                    CASE
                        WHEN Target_Disease = :dom THEN 0
                        WHEN Target_Disease = 'General Wellness' THEN 1
                        ELSE 2
                    END,
                    -- Priority 2: Randomize the results so they get a fresh 4 every day!
                    NEWID()
            """)

        rows = db.session.execute(
            sql,
            {
                "dom": dominant,
                "sev": severity,
                "maxd": int(max_duration),
            },
        ).fetchall()

        remedies = []
        for r in rows:
            remedies.append({
                "Remedy_ID": int(r[0]),
                "Title": r[1],
                "Description": r[2],
                "Remedy_Type": r[3] or "General",
                "Duration_Minutes": int(r[4]) if r[4] is not None else 10,
                "Target_Severity": r[5],
                "Resource_Link": r[6],
                "Linked_Sub_Category_ID": r[7],
            })

        return remedies

    except Exception as e:
        print("Remedy library fetch failed:", e)
        # Return empty list safely. NEVER return fake IDs.
        return []


def group_remedies_by_duration(remedies):
    grouped = {
        "immediate_0_5_min": [],
        "quick_6_15_min": [],
        "standard_16_30_min": [],
        "extended_30_plus": [],
    }

    for r in remedies:
        mins = int(r.get("Duration_Minutes") or 0)
        if mins <= 5:
            grouped["immediate_0_5_min"].append(r)
        elif mins <= 15:
            grouped["quick_6_15_min"].append(r)
        elif mins <= 30:
            grouped["standard_16_30_min"].append(r)
        else:
            grouped["extended_30_plus"].append(r)

    return grouped


def build_modality_distribution(remedies):
    out = {}
    for r in remedies:
        rtype = r.get("Remedy_Type") or "General"
        out[rtype] = out.get(rtype, 0) + 1
    return out

@app.post("/api/signup")
def student_signup():
    """Handles new student registration"""
    data = request.get_json()

    name = data.get("name")
    email = data.get("email")
    gender = data.get("gender")
    arid_no = data.get("arid_no")
    semester = data.get("semester")
    password = data.get("password")

    if not all([name, email, gender, arid_no, semester, password]):
        return jsonify({"ok": False, "message": "All fields are required"}), 400

    try:
        # 1. Check if ARID No or Email already exists
        existing = db.session.execute(
            text("SELECT 1 FROM dbo.Students WHERE Arid_No = :arid OR Email = :email"),
            {"arid": arid_no, "email": email}
        ).fetchone()

        if existing:
            return jsonify({"ok": False, "message": "A student with this ARID number or Email already exists!"}), 400

        # 2. Automatically assign them to an Advisor (picks a random active advisor)
        advisor = db.session.execute(text("SELECT TOP 1 Advisor_ID FROM dbo.Advisors ORDER BY NEWID()")).fetchone()
        adv_id = advisor[0] if advisor else None

        # 3. Save the new student
        db.session.execute(
            text("""
                INSERT INTO dbo.Students (Arid_No, Student_Name, Email, Gender, Semester, Password, Advisor_ID)
                VALUES (:arid, :name, :email, :gender, :semester, :pass, :adv_id)
            """),
            {
                "arid": arid_no,
                "name": name,
                "email": email,
                "gender": gender,
                "semester": semester,
                "pass": password,
                "adv_id": adv_id
            }
        )
        db.session.commit()

        return jsonify({"ok": True, "message": "Account created successfully! You can now log in."})
    except Exception as e:
        db.session.rollback()
        print(f"Signup error: {e}")
        return jsonify({"ok": False, "message": "Server error during signup."}), 500

@app.post("/api/login")
def login():
    """Handles login for Students, Advisors, and Psychologists"""
    data = request.get_json()

    if not data:
        return jsonify({"ok": False, "message": "No data provided"}), 400

    # Safely grab the input whether the frontend calls it 'login_id' or 'arid_no'
    login_input = data.get("login_id") or data.get("arid_no")
    password_input = data.get("password")

    if not login_input or not password_input:
        return jsonify({"ok": False, "message": "Missing credentials"}), 400

    try:
        # 1. STUDENT CHECK
        student = db.session.execute(
            text(
                "SELECT Student_ID, Student_Name, Arid_No FROM dbo.Students WHERE Arid_No = :arid AND Password = :pass"),
            {"arid": login_input, "pass": password_input}
        ).fetchone()

        if student:
            return jsonify({"ok": True, "role": "student", "student_id": student[0], "name": student[1],
                            "arid_no": student[2]}), 200

        # 2. ADVISOR CHECK
        advisor = db.session.execute(
            text("SELECT Advisor_ID, Advisor_Name FROM dbo.Advisors WHERE Advisor_Code = :code AND Password = :pass"),
            {"code": login_input, "pass": password_input}
        ).fetchone()

        if advisor:
            return jsonify({"ok": True, "role": "advisor", "advisor_id": advisor[0], "name": advisor[1]}), 200

        # 3. PSYCHOLOGIST CHECK
        psychologist = db.session.execute(
            text(
                "SELECT Psychologist_ID, Name FROM dbo.Psychologists WHERE Psychologist_Code = :code AND Password = :pass"),
            {"code": login_input, "pass": password_input}
        ).fetchone()

        if psychologist:
            return jsonify(
                {"ok": True, "role": "psychologist", "psychologist_id": psychologist[0], "name": psychologist[1]}), 200

        # 4. ADMIN CHECK
        admin = db.session.execute(
            text("SELECT Admin_ID, Name FROM dbo.Admins WHERE Admin_Code = :code AND Password = :pass"),
            {"code": login_input, "pass": password_input}
        ).fetchone()

        if admin:
            return jsonify({"ok": True, "role": "admin", "admin_id": admin[0], "name": admin[1]}), 200

        # 5. NO MATCH FOUND
        return jsonify({"ok": False, "message": "Invalid Login ID or Password"}), 401

    except Exception as e:
        print(f"Login error: {e}")
        return jsonify({"ok": False, "message": "Server error during login"}), 500

@app.get("/api/advisor/alerts-and-meetings")
def get_advisor_alerts():
    """Fetches new high-alert students needing meetings, and students who accepted meetings"""
    adv_id = request.args.get("advisor_id")
    if not adv_id: return jsonify({"ok": False}), 400

    try:
        # 1. New Alerts: High severity students who do NOT have an active referral OR active advisor meeting
        new_alerts = db.session.execute(text("""
            SELECT DISTINCT s.Student_ID, s.Student_Name, s.Arid_No
            FROM dbo.Students s
            WHERE s.Advisor_ID = :aid AND (
                EXISTS (SELECT 1 FROM dbo.General_Test_Result g WHERE g.Student_ID = s.Student_ID AND (g.Severity_Label LIKE '%High%' OR g.Severity_Label LIKE '%Severe%'))
                OR EXISTS (SELECT 1 FROM dbo.Weekly_CheckIn_Session w WHERE w.Student_ID = s.Student_ID AND (w.Triggered_Disease LIKE '%SH%' OR w.Triggered_Category_Key LIKE '%SH%' OR w.Total_Score >= 15))
                OR EXISTS (
                    SELECT 1 FROM dbo.Test_Session ts
                    JOIN dbo.Student_Test_Answer sta ON sta.Test_ID = ts.Test_ID
                    JOIN dbo.Category_Gates cg ON cg.Gate_ID = sta.Ref_ID
                    JOIN dbo.Categories c ON c.Category_ID = cg.Category_ID
                    WHERE ts.Student_ID = s.Student_ID AND sta.Item_Type = 'Gate' AND c.Category_Key = 'SH' AND sta.Answer_Value >= 1
                )
            )

@app.post("/api/save-profile")
def save_profile():
    data = request.get_json(force=True) or {}
    arid_no = (data.get("Arid_No") or "").strip()
    responses = data.get("responses") or []

    if not arid_no:
        return jsonify({"ok": False, "message": "Missing Arid_No"}), 400

    sid = get_student_id_by_arid(arid_no)
    if not sid:
        return jsonify({"ok": False, "message": "Student not found"}), 404

    try:
        for item in responses:
            qid = item.get("question_id")
            ans = item.get("answer")
            if qid is None:
                continue

            db.session.execute(
                text("""
                        MERGE dbo.Student_Profile_Response AS target
                        USING (VALUES (:sid, :qid, :ans))
                              AS source (Student_ID, Profile_Question_ID, Response_Value)
                        ON target.Student_ID = source.Student_ID
                           AND target.Profile_Question_ID = source.Profile_Question_ID
                        WHEN MATCHED THEN
                            UPDATE SET
                                Response_Value = source.Response_Value,
                                Created_On = SYSUTCDATETIME()
                        WHEN NOT MATCHED THEN
                            INSERT (Student_ID, Profile_Question_ID, Response_Value)
                            VALUES (source.Student_ID, source.Profile_Question_ID, source.Response_Value);
                    """),
                {"sid": sid, "qid": int(qid), "ans": str(ans) if ans is not None else None},
            )

        db.session.commit()
        return jsonify({"ok": True, "message": "Profile saved successfully"})
    except Exception as e:
        db.session.rollback()
        return jsonify({"ok": False, "message": str(e)}), 500

        return jsonify({"ok": False, "message": "Missing arid_no"}), 400

    sid = get_student_id_by_arid(arid_no)

    live_with_family = (get_profile_value(sid, "live_with_family") or "").strip()
    mother_status = (get_profile_value(sid, "mother_status") or "").strip()
    father_status = (get_profile_value(sid, "father_status") or "").strip()

    allow_ct = (live_with_family.lower() == "yes")
    allow_fd = (mother_status.lower() == "alive") or (father_status.lower() == "alive")

    rows = db.session.execute(
        text("""
                SELECT g.Gate_ID, c.Category_Key, c.Category_Name, g.Gate_Text, g.Sort_Order
                FROM dbo.Category_Gates g
                JOIN dbo.Categories c ON c.Category_ID = g.Category_ID
                WHERE g.Is_Active = 1
                  AND c.Is_Active = 1
                ORDER BY g.Sort_Order ASC
            """)
    ).fetchall()

    filtered = []
    for r in rows:
        gate_id = int(r[0])
        cat_key = r[1]
        if cat_key == "CT" and not allow_ct:
            continue
        if cat_key == "FD" and not allow_fd:
            continue

        filtered.append({
            "id": gate_id,
            "category_key": cat_key,
            "category_name": r[2],
            "text": r[3],
        })

    return jsonify(filtered)

@app.get("/api/student/meeting-status")
def student_meeting_status():
    arid_no = request.args.get("arid_no")
    if not arid_no: return jsonify({"ok": False}), 400

    try:
        student = db.session.execute(text("SELECT Student_ID FROM dbo.Students WHERE Arid_No = :arid"),
                                     {"arid": arid_no}).fetchone()
        if not student: return jsonify({"ok": False}), 404

        meeting = db.session.execute(
            text(
                "SELECT Status, Meeting_Date FROM dbo.Psychologist_Meetings WHERE Student_ID = :sid ORDER BY Requested_On DESC"),
            {"sid": student[0]}
        ).fetchone()

        referred = db.session.execute(
            text("SELECT 1 FROM dbo.Student_Referrals WHERE Student_ID = :sid AND Status = 'Pending'"),
            {"sid": student[0]}
        ).fetchone()

        needs_meeting = False
        already_requested = False
        is_scheduled = False
        meeting_date_str = None

        if meeting:
            # 🔥 BULLETPROOF FIX: .strip() removes invisible spaces from SQL
            status_text = meeting[0].strip() if meeting[0] else ""

            if status_text == 'Requested':
                already_requested = True
            elif status_text == 'Scheduled':
                is_scheduled = True
                if meeting[1]:
                    d = meeting[1]
                    meeting_date_str = d.strftime('%b %d, %Y - %I:%M %p') if hasattr(d, 'strftime') else str(d)

        if referred and not already_requested and not is_scheduled:
            needs_meeting = True

        return jsonify({
            "ok": True,
            "needs_meeting": needs_meeting,
            "already_requested": already_requested,
            "is_scheduled": is_scheduled,
            "meeting_date": meeting_date_str
        })
    except Exception as e:
        print(f"Student meeting status error: {e}")
        return jsonify({"ok": False}), 500

@app.post("/api/psychologist/schedule-meeting")
def schedule_psy_meeting():
    data = request.get_json()
    meeting_id = data.get("meeting_id")
    meeting_date = data.get("meeting_date")
    notes = data.get("notes", "")
    student_arid = data.get("student_arid")  # For new direct scheduling

    try:
        clean_date = meeting_date.replace('T', ' ') if meeting_date else None

        if meeting_id:
            # Existing flow: update requested meeting
            db.session.execute(text(
                "UPDATE dbo.Psychologist_Meetings SET Status = 'Scheduled', Meeting_Date = :mdate, Notes = :notes WHERE Meeting_ID = :mid"),
                {"mdate": clean_date, "notes": notes, "mid": meeting_id}
            )
        elif student_arid:
            # New direct scheduling flow
            student = db.session.execute(
                text("SELECT Student_ID FROM dbo.Students WHERE Arid_No = :arid"),
                {"arid": student_arid}
            ).fetchone()

            if not student:
                return jsonify({"ok": False, "message": "Student not found"}), 404

            # Check for double-booking
            existing = db.session.execute(
                text("""
                    SELECT COUNT(*) FROM dbo.Psychologist_Meetings 
                    WHERE Status = 'Scheduled' 
                    AND Meeting_Date IS NOT NULL
                    AND ABS(DATEDIFF(MINUTE, Meeting_Date, :mdate)) < 60
                """),
                {"mdate": clean_date}
            ).scalar()

            if existing and existing > 0:
                return jsonify(
                    {"ok": False, "message": "This time slot is already booked. Please choose another."}), 409

            db.session.execute(
                text("""
                    INSERT INTO dbo.Psychologist_Meetings (Student_ID, Status, Meeting_Date, Notes, Requested_On)
                    VALUES (:sid, 'Scheduled', :mdate, :notes, SYSUTCDATETIME())
                """),
                {"sid": student[0], "mdate": clean_date, "notes": notes}
            )
        else:
            return jsonify({"ok": False, "message": "Missing meeting_id or student_arid"}), 400

        db.session.commit()
        return jsonify({"ok": True})
    except Exception as e:

@app.post("/api/test/deepdive/init")
def deepdive_init():
    data = request.get_json(force=True) or {}
    test_id = int(data.get("test_id") or 0)
    if not test_id:
        return jsonify({"ok": False, "message": "Missing test_id"}), 400

    existing = db.session.execute(
        text("SELECT COUNT(*) FROM dbo.Test_Question_Queue WHERE Test_ID = :tid"),
        {"tid": test_id},
    ).scalar() or 0

    if int(existing) > 0:
        return get_next_question_internal(test_id)

    state_exists = db.session.execute(
        text("SELECT COUNT(*) FROM dbo.Test_Category_State WHERE Test_ID = :tid"),
        {"tid": test_id},
    ).scalar() or 0

    if int(state_exists) == 0:
        rows = db.session.execute(
            text("""
                    SELECT DISTINCT c.Category_ID, c.Category_Key
                    FROM dbo.Student_Test_Answer a
                    JOIN dbo.Category_Gates g ON g.Gate_ID = a.Ref_ID
                    JOIN dbo.Categories c ON c.Category_ID = g.Category_ID
                    WHERE a.Test_ID = :tid
                      AND a.Item_Type = 'Gate'
                      AND (
                            (c.Category_Key = 'SH' AND a.Answer_Value >= 1)
                            OR (c.Category_Key != 'SH' AND a.Answer_Value >= 3)
                          )
                """),
            {"tid": test_id},
        ).fetchall()

        for (cat_id, _) in rows:
            db.session.execute(
                text("""
                        INSERT INTO dbo.Test_Category_State
                        (Test_ID, Category_ID, Gate_Answer, Current_Base_Order, Is_Completed)
                        VALUES (:tid, :cid, 1, 0, 0)
                    """),
                {"tid": test_id, "cid": cat_id},
            )
        db.session.commit()

    category_row = db.session.execute(
        text("""
                SELECT TOP 1 tcs.Category_ID, tcs.Current_Base_Order, c.Category_Key
                FROM dbo.Test_Category_State tcs
                JOIN dbo.Categories c ON c.Category_ID = tcs.Category_ID
                WHERE tcs.Test_ID = :tid
                  AND tcs.Is_Completed = 0
                ORDER BY tcs.Category_ID
            """),
        {"tid": test_id},
    ).fetchone()

    if not category_row:
        return jsonify({"ok": False, "message": "No triggered categories found"}), 400

    cat_id, _, _ = category_row

    base_row = db.session.execute(
        text("""
                SELECT TOP 1 Base_ID, Base_Order
                FROM dbo.DeepDive_Base_Questions
                WHERE Category_ID = :cid
                  AND Is_Active = 1
                ORDER BY Base_Order
            """),
        {"cid": cat_id},
    ).fetchone()

    if not base_row:
        return jsonify({"ok": False, "message": "No base questions for category"}), 400

    base_id, base_order = base_row

    db.session.execute(
        text("""
                INSERT INTO dbo.Test_Question_Queue
                (Test_ID, Item_Type, Ref_ID, Order_Index, Status, Parent_Category_ID)
                VALUES (:tid, 'Base', :rid, 1, 'Pending', :cid)
            """),
        {"tid": test_id, "rid": base_id, "cid": cat_id},
    )

    db.session.execute(
        text("""
                UPDATE dbo.Test_Category_State
                SET Current_Base_Order = :bo,
                    Updated_On = SYSUTCDATETIME()
                WHERE Test_ID = :tid
                  AND Category_ID = :cid
            """),
        {"tid": test_id, "cid": cat_id, "bo": base_order},
    )

    db.session.commit()
    return get_next_question_internal(test_id)


def get_next_question_internal(test_id):
    row = db.session.execute(
        text("""
                SELECT TOP 1 Queue_ID, Item_Type, Ref_ID, Parent_Category_ID
                FROM dbo.Test_Question_Queue
                WHERE Test_ID = :tid
                  AND Status = 'Pending'
                ORDER BY Order_Index ASC, Queue_ID ASC
            """),
        {"tid": test_id},
    ).fetchone()

    if not row:
        incomplete = db.session.execute(
            text("""
                    SELECT COUNT(*)
                    FROM dbo.Test_Category_State
                    WHERE Test_ID = :tid
                      AND Is_Completed = 0
                """),
            {"tid": test_id},
        ).scalar() or 0

        if int(incomplete) == 0:
            return jsonify({"ok": True, "done": True, "message": "Assessment complete"})

        return advance_to_next_category(test_id)

    queue_id, item_type, ref_id, _ = row

    if item_type == "Base":
        q = db.session.execute(
            text("""
                    SELECT b.Base_ID, b.Base_Text, c.Category_Key, c.Category_Name
                    FROM dbo.DeepDive_Base_Questions b
                    JOIN dbo.Categories c ON c.Category_ID = b.Category_ID
                    WHERE b.Base_ID = :id
                """),
            {"id": ref_id},
        ).fetchone()
    else:
        q = db.session.execute(
            text("""
                    SELECT s.Sub_ID, s.Sub_Text, c.Category_Key, c.Category_Name
                    FROM dbo.DeepDive_Sub_Questions s
                    JOIN dbo.DeepDive_Base_Questions b ON b.Base_ID = s.Base_ID
                    JOIN dbo.Categories c ON c.Category_ID = b.Category_ID
                    WHERE s.Sub_ID = :id
                """),
            {"id": ref_id},
        ).fetchone()

    return jsonify({
        "ok": True,
        "queue_id": queue_id,
        "item_type": item_type,
        "question_text": q[1],
        "category_key": q[2],
        "category_name": q[3],
        "options": LIKERT_5,
        "is_conversational": True,
    })


def advance_to_next_category(test_id):
    next_cat = db.session.execute(
        text("""
                SELECT TOP 1 tcs.Category_ID, c.Category_Key
                FROM dbo.Test_Category_State tcs
                JOIN dbo.Categories c ON c.Category_ID = tcs.Category_ID
                WHERE tcs.Test_ID = :tid
                  AND tcs.Is_Completed = 0
                ORDER BY tcs.Category_ID
            """),
        {"tid": test_id},
    ).fetchone()

    if not next_cat:
        return jsonify({"ok": True, "done": True, "message": "All categories complete"})

    cat_id, _ = next_cat

    base_row = db.session.execute(
        text("""
                SELECT TOP 1 Base_ID, Base_Order
                FROM dbo.DeepDive_Base_Questions
                WHERE Category_ID = :cid
                  AND Is_Active = 1
                ORDER BY Base_Order
            """),
        {"cid": cat_id},
    ).fetchone()

    if not base_row:
        db.session.execute(
            text("""
                    UPDATE dbo.Test_Category_State
                    SET Is_Completed = 1,
                        Updated_On = SYSUTCDATETIME()
                    WHERE Test_ID = :tid
                      AND Category_ID = :cid
                """),
            {"tid": test_id, "cid": cat_id},
        )
        db.session.commit()
        return advance_to_next_category(test_id)

    base_id, base_order = base_row

    db.session.execute(
        text("""
                INSERT INTO dbo.Test_Question_Queue
                (Test_ID, Item_Type, Ref_ID, Order_Index, Status, Parent_Category_ID)
                VALUES (:tid, 'Base', :rid, 1, 'Pending', :cid)
            """),
        {"tid": test_id, "rid": base_id, "cid": cat_id},
    )

    db.session.execute(
        text("""
                UPDATE dbo.Test_Category_State
                SET Current_Base_Order = :bo,
                    Updated_On = SYSUTCDATETIME()
                WHERE Test_ID = :tid
                  AND Category_ID = :cid
            """),
        {"tid": test_id, "cid": cat_id, "bo": base_order},
    )

    db.session.commit()
    return get_next_question_internal(test_id)

@app.post("/api/test/answer")
def test_answer():
    data = request.get_json(force=True) or {}

    test_id = data.get("test_id")
    queue_id = data.get("queue_id")
    score_raw = data.get("score")

    if not test_id or not queue_id or score_raw is None:
        return jsonify({"ok": False, "message": "Missing test_id, queue_id, or score"}), 400

    try:
        test_id = int(test_id)
        queue_id = int(queue_id)
        score = int(score_raw)
    except Exception:
        return jsonify({"ok": False, "message": "Invalid test_id/queue_id/score"}), 400

    status_row = db.session.execute(
        text("""
                SELECT Status
                FROM dbo.Test_Question_Queue
                WHERE Queue_ID = :qid
                  AND Test_ID = :tid
            """),
        {"qid": queue_id, "tid": test_id},
    ).fetchone()

    if not status_row or (status_row[0] or "").lower() != "pending":
        return get_next_question_internal(test_id)

    row = db.session.execute(
        text("""
                SELECT
                    q.Item_Type,
                    q.Ref_ID,
                    COALESCE(b.Category_ID, parent_b.Category_ID) AS Category_ID
                FROM dbo.Test_Question_Queue q WITH (UPDLOCK, ROWLOCK)
                LEFT JOIN dbo.DeepDive_Base_Questions b
                    ON b.Base_ID = q.Ref_ID AND q.Item_Type = 'Base'
                LEFT JOIN dbo.DeepDive_Sub_Questions s
                    ON s.Sub_ID = q.Ref_ID AND q.Item_Type = 'Sub'
                LEFT JOIN dbo.DeepDive_Base_Questions parent_b
                    ON parent_b.Base_ID = s.Base_ID
                WHERE q.Queue_ID = :qid
                  AND q.Test_ID = :tid
            """),
        {"qid": queue_id, "tid": test_id},
    ).fetchone()

    if not row:
        db.session.commit()
        return get_next_question_internal(test_id)

    item_type, ref_id, category_id = row
    ref_id = int(ref_id)
    category_id = int(category_id) if category_id else None

    db.session.execute(
        text("""
                INSERT INTO dbo.Student_Test_Answer (Test_ID, Item_Type, Ref_ID, Answer_Value)
                VALUES (:tid, :t, :rid, :val)
            """),
        {"tid": test_id, "t": item_type, "rid": ref_id, "val": score},
    )

    db.session.execute(
        text("""
                UPDATE dbo.Test_Question_Queue
                SET Status = 'Answered'
                WHERE Queue_ID = :qid
                  AND Test_ID = :tid
            """),
        {"qid": queue_id, "tid": test_id},
    )

    if item_type == "Base":
        threshold_info = db.session.execute(
            text("""
                    SELECT c.Category_Key, c.Base_Trigger_Min_Value, b.Trigger_Min_Value, b.Base_Order
                    FROM dbo.DeepDive_Base_Questions b
                    JOIN dbo.Categories c ON c.Category_ID = b.Category_ID
                    WHERE b.Base_ID = :bid
                """),
            {"bid": ref_id},
        ).fetchone()

        # Python now trusts the database's Trigger_Min_Value completely!
        effective_threshold = threshold_info[2] if threshold_info[2] is not None else threshold_info[1]
        current_base_order = int(threshold_info[3] or 0)

        if score >= int(effective_threshold or 0):
            subs = db.session.execute(
                text("""
                        SELECT Sub_ID
                        FROM dbo.DeepDive_Sub_Questions
                        WHERE Base_ID = :bid
                          AND Is_Active = 1
                        ORDER BY Sub_Order
                    """),
                {"bid": ref_id},
            ).fetchall()

            if subs:
                for idx, (sub_id,) in enumerate(subs, start=2):
                    db.session.execute(
                        text("""
                                INSERT INTO dbo.Test_Question_Queue
                                (Test_ID, Item_Type, Ref_ID, Order_Index, Status)
                                VALUES (:tid, 'Sub', :rid, :ord, 'Pending')
                            """),
                        {"tid": test_id, "rid": int(sub_id), "ord": idx},
                    )
            else:
                insert_next_base(test_id, category_id, current_base_order)
        else:
            insert_next_base(test_id, category_id, current_base_order)

    elif item_type == "Sub":
        more_pending = db.session.execute(
            text("""
                    SELECT COUNT(*)
                    FROM dbo.Test_Question_Queue
                    WHERE Test_ID = :tid
                      AND Status = 'Pending'
                """),
            {"tid": test_id},
        ).scalar() or 0

        if int(more_pending) == 0:
            state_row = db.session.execute(
                text("""
                        SELECT Current_Base_Order
                        FROM dbo.Test_Category_State
                        WHERE Test_ID = :tid
                          AND Category_ID = :cid
                    """),
                {"tid": test_id, "cid": category_id},
            ).fetchone()
            current_bo = int(state_row[0]) if state_row and state_row[0] is not None else 0
            insert_next_base(test_id, category_id, current_bo)

    db.session.commit()
    return jsonify({"ok": True})


def insert_next_base(test_id, category_id, current_base_order):
    any_pending = db.session.execute(
        text("""
                SELECT COUNT(*)
                FROM dbo.Test_Question_Queue
                WHERE Test_ID = :tid
                  AND Status = 'Pending'
            """),
        {"tid": test_id},
    ).scalar() or 0

    if int(any_pending) > 0:
        return

    next_base = db.session.execute(
        text("""
                SELECT TOP 1 Base_ID, Base_Order
                FROM dbo.DeepDive_Base_Questions
                WHERE Category_ID = :cid
                  AND Base_Order > :current_order
                  AND Is_Active = 1
                ORDER BY Base_Order
            """),
        {"cid": category_id, "current_order": current_base_order},
    ).fetchone()

    if next_base:
        base_id, base_order = next_base

        max_order = db.session.execute(
            text("""
                    SELECT COALESCE(MAX(Order_Index), 0)
                    FROM dbo.Test_Question_Queue
                    WHERE Test_ID = :tid
                """),
            {"tid": test_id},
        ).scalar() or 0

        new_order = int(max_order) + 10

        db.session.execute(
            text("""
                    INSERT INTO dbo.Test_Question_Queue
                    (Test_ID, Item_Type, Ref_ID, Order_Index, Status, Parent_Category_ID)
                    VALUES (:tid, 'Base', :rid, :ord, 'Pending', :cid)
                """),
            {"tid": test_id, "rid": base_id, "ord": new_order, "cid": category_id},
        )

        db.session.execute(
            text("""
                    UPDATE dbo.Test_Category_State
                    SET Current_Base_Order = :bo,
                        Updated_On = SYSUTCDATETIME()
                    WHERE Test_ID = :tid
                      AND Category_ID = :cid
                """),
            {"tid": test_id, "cid": category_id, "bo": base_order},
        )
    else:
        db.session.execute(
            text("""
                    UPDATE dbo.Test_Category_State
                    SET Is_Completed = 1,
                        Updated_On = SYSUTCDATETIME()
                    WHERE Test_ID = :tid
                      AND Category_ID = :cid
                """),

@app.get("/api/advisor/booked-slots")

    """
    Returns all upcoming accepted/proposed advisor meeting slots
    so the advisor can see conflicts before proposing a new time.
    """
    advisor_id = request.args.get("advisor_id")
    if not advisor_id:
        return jsonify({"ok": False, "message": "Missing advisor_id"}), 400

    try:
        rows = db.session.execute(
            text("""
                SELECT
                    s.Student_Name,
                    m.Proposed_Slots,
                    m.Confirmed_Slot,
                    m.Status
                FROM dbo.Advisor_Student_Meetings m
                INNER JOIN dbo.Students s ON m.Student_ID = s.Student_ID
                WHERE m.Advisor_ID = :aid
                  AND m.Status IN ('Proposed', 'Accepted')
                  AND (
                      m.Proposed_Slots IS NOT NULL
                      OR m.Confirmed_Slot IS NOT NULL
                  )
                ORDER BY m.Proposed_Slots DESC
            """),

        ).fetchall()

                "student_name": r[0],
                "datetime": dt_str,  # ISO string or datetime string

            })

        ).fetchone()

            {"aid": adv_id}

            text("""
                    SELECT DISTINCT s.Student_ID, s.Student_Name, s.Arid_No

                    WHERE s.Advisor_ID = :aid AND (

                        -- CONDITION 1: General Test Result has High/Severe severity
                        EXISTS (
                            SELECT 1 FROM dbo.General_Test_Result g 
                            WHERE g.Student_ID = s.Student_ID 
                            AND (g.Severity_Label LIKE '%High%' OR g.Severity_Label LIKE '%Severe%')

                        -- CONDITION 2: Weekly Check-In triggered SH or high score

                                w.Triggered_Disease LIKE '%SH%' OR 
                                w.Triggered_Disease LIKE '%Self%' OR
                                w.Triggered_Category_Key LIKE '%SH%' OR 
                                w.Triggered_Category_Key LIKE '%Self%' OR
                                w.Total_Score >= 15

                            FROM dbo.Test_Session ts
                            JOIN dbo.Student_Test_Answer sta ON sta.Test_ID = ts.Test_ID
                            JOIN dbo.Category_Gates cg ON cg.Gate_ID = sta.Ref_ID

@app.post("/api/advisor/refer-student")
def refer_student_to_psychologist():
    """
    Refers a student to a psychologist.
    - Cleans up any stuck/broken referral rows (NULL or unknown status) first.
    - If student was previously referred and had a completed meeting → re-refer same psychologist.
    - First time → assign least busy psychologist.
    """
    data = request.get_json()
    arid_no = data.get("arid_no")
    adv_id = data.get("advisor_id")

    if not arid_no or not adv_id:
        return jsonify({"ok": False, "message": "Missing data"}), 400

    try:
        # 1. Get Student ID
        student = db.session.execute(
            text("SELECT Student_ID FROM dbo.Students WHERE Arid_No = :arid"),
            {"arid": arid_no}
        ).fetchone()

        if not student:
            return jsonify({"ok": False, "message": "Student not found"}), 404

        sid = student[0]

        # 2. FIX: Clean up any stuck rows with NULL or unknown status before checking.
        # These are leftovers from old broken inserts. Delete them so they don't
        # permanently block new referrals.
        db.session.execute(
            text("""
                DELETE FROM dbo.Student_Referrals
                WHERE Student_ID = :sid
                  AND (Status IS NULL OR Status NOT IN ('Pending', 'Meeting_Requested', 'Completed', 'Cancelled'))
            """),
            {"sid": sid}
        )
        db.session.commit()

        # 3. Now check for a genuinely active referral (Pending or Meeting_Requested)
        already_referred = db.session.execute(
            text("""
                SELECT Referral_ID, Psychologist_ID
                FROM dbo.Student_Referrals
                WHERE Student_ID = :sid
                AND Psychologist_ID IS NOT NULL
                AND Status IN ('Pending', 'Meeting_Requested')
            """),
            {"sid": sid}
        ).fetchone()

        if already_referred:
            # Return already_referred=True so the frontend knows this is NOT a new referral
            return jsonify({
                "ok": True,
                "already_referred": True,
                "message": "Student already has an active referral pending with the psychologist."
            }), 200

        # 4. Check if student has a previous COMPLETED referral → re-refer to same psychologist
        previous = db.session.execute(
            text("""
                SELECT TOP 1 sr.Psychologist_ID, p.Name
                FROM dbo.Student_Referrals sr
                INNER JOIN dbo.Psychologists p ON sr.Psychologist_ID = p.Psychologist_ID
                WHERE sr.Student_ID = :sid
                  AND sr.Status = 'Completed'
                  AND sr.Psychologist_ID IS NOT NULL
                ORDER BY sr.Referred_On DESC
            """),
            {"sid": sid}
        ).fetchone()

        if previous:
            assigned_psy_id = previous[0]
            assigned_psy_name = previous[1]
            note = f"Re-referred to same psychologist: {assigned_psy_name}"
        else:
            # First time — pick least busy psychologist by active referral count
            least_busy = db.session.execute(
                text("""
                    SELECT TOP 1 p.Psychologist_ID, p.Name
                    FROM dbo.Psychologists p
                    LEFT JOIN dbo.Student_Referrals sr
                        ON p.Psychologist_ID = sr.Psychologist_ID
                        AND sr.Status IN ('Pending', 'Meeting_Requested')
                    GROUP BY p.Psychologist_ID, p.Name
                    ORDER BY COUNT(sr.Student_ID) ASC
                """)
            ).fetchone()

            if not least_busy:
                return jsonify({"ok": False, "message": "No psychologists available in the system."}), 500

            assigned_psy_id = least_busy[0]
            assigned_psy_name = least_busy[1]
            note = f"First referral — assigned to: {assigned_psy_name}"

        # 5. Insert with explicit Status='Pending' and Referred_On timestamp
        db.session.execute(
            text("""
                INSERT INTO dbo.Student_Referrals
                (
                    Student_ID,
                    Advisor_ID,
                    Psychologist_ID,
                    Status,
                    Referred_On
                )
                VALUES
                (
                    :sid,
                    :aid,
                    :pid,
                    'Pending',
                    SYSUTCDATETIME()
                )
            """),
            {
                "sid": sid,
                "aid": adv_id,
                "pid": assigned_psy_id
            }
        )

        db.session.commit()

        print(f"[REFERRAL] Student {arid_no} → Psychologist {assigned_psy_name} (ID:{assigned_psy_id}) | {note}")
        return jsonify({
            "ok": True,
            "already_referred": False,
            "message": "Referral successful",
            "psychologist_name": assigned_psy_name,
            "note": note
        }), 200

    except Exception as e:
        db.session.rollback()
        print(f"[REFERRAL ERROR] {e}")
        return jsonify({"ok": False, "message": f"Server error: {str(e)}"}), 500

@app.get("/api/remedies/history")
def remedies_history():
    arid_no = (request.args.get("arid_no") or "").strip()
    if not arid_no:
        return jsonify({"ok": False, "message": "Missing arid_no"}), 400

    sid = get_student_id_by_arid(arid_no)
    if not sid:
        return jsonify({"ok": False, "message": "Student not found"}), 404

    try:
        # Check if table exists
        rows = db.session.execute(
            text("""
                    SELECT
                        h.History_ID,
                        h.Test_ID,
                        h.Remedy_ID,
                        h.Status,
                        h.Feedback_Rating as Rating,  -- Note: column name may differ
                        h.Feedback_Text,
                        h.Suggested_On as Created_On,
                        h.Completed_On,
                        h.Viewed_On,
                        h.Dismissed_On,
                        r.Title,
                        r.Remedy_Type as Type,
                        r.Duration_Minutes,
                        r.Resource_Link
                    FROM dbo.Student_Remedy_History h
                    LEFT JOIN dbo.Remedy_Library r ON r.Remedy_ID = h.Remedy_ID
                    WHERE h.Student_ID = :sid
                    ORDER BY h.Suggested_On DESC
                """),
            {"sid": sid},
        ).fetchall()

        # Organize into toolkit format expected by frontend
        toolkit = {
            "completed": [],
            "saved": [],
            "rated": [],
            "dismissed": []
        }

        total_suggested = 0

        for r in rows:
            status = r[3]
            item = {
                "history_id": int(r[0]) if r[0] else None,
                "test_id": int(r[1]) if r[1] else None,
                "remedy_id": int(r[2]) if r[2] else None,
                "status": status,
                "rating": int(r[4]) if r[4] else None,
                "feedback_text": r[5],
                "suggested_on": str(r[6]) if r[6] else None,
                "completed_on": str(r[7]) if r[7] else None,
                "title": r[10] or "Unknown Remedy",
                "type": r[11] or "General",
                "duration_minutes": int(r[12]) if r[12] else 10,
                "resource_link": r[13]
            }

            total_suggested += 1

            # Categorize for tabs
            if status == "Completed":
                toolkit["completed"].append(item)
                if item["rating"]:
                    toolkit["rated"].append(item)
            elif status == "Saved":
                toolkit["saved"].append(item)
            elif status == "Dismissed":
                toolkit["dismissed"].append(item)
            elif status in ["Suggested", "Viewed"]:
                toolkit["saved"].append(item)  # Active remedies go to saved tab

        completed_count = len(toolkit["completed"])
        completion_rate = (completed_count / total_suggested * 100) if total_suggested > 0 else 0

        return jsonify({
            "ok": True,
            "toolkit": toolkit,
            "total_remedies_suggested": total_suggested,
            "completion_rate": round(completion_rate, 1)
        })

    except Exception as e:
        print("Remedy history error:", e)
        # Return empty structure on error so UI doesn't crash
        return jsonify({
            "ok": True,  # Return ok=True to prevent UI error state
            "toolkit": {"completed": [], "saved": [], "rated": [], "dismissed": []},
            "total_remedies_suggested": 0,
            "completion_rate": 0,
            "message": "History table may not exist or schema mismatch"
        })

@app.get("/api/test/details")
def test_details():
    """Fetches all questions and answers for a specific test session"""
    test_id = request.args.get("test_id")
    if not test_id:
        return jsonify({"ok": False, "message": "Missing test_id"}), 400

    try:
        test_id = int(test_id)

        # Helper dictionary to convert numbers to text answers
        likert_map = {0: "Never", 1: "Rarely", 2: "Sometimes", 3: "Often", 4: "Always"}

        # 1. Fetch Gate Questions
        gates = db.session.execute(
            text("""
                    SELECT c.Category_Name, cg.Gate_Text, a.Answer_Value
                    FROM dbo.Student_Test_Answer a
                    JOIN dbo.Category_Gates cg ON cg.Gate_ID = a.Ref_ID
                    JOIN dbo.Categories c ON c.Category_ID = cg.Category_ID
                    WHERE a.Test_ID = :tid AND a.Item_Type = 'Gate'
                """), {"tid": test_id}
        ).fetchall()

        # 2. Fetch Base DeepDive Questions
        bases = db.session.execute(
            text("""
                    SELECT c.Category_Name, b.Base_Text, a.Answer_Value
                    FROM dbo.Student_Test_Answer a
                    JOIN dbo.DeepDive_Base_Questions b ON b.Base_ID = a.Ref_ID
                    JOIN dbo.Categories c ON c.Category_ID = b.Category_ID
                    WHERE a.Test_ID = :tid AND a.Item_Type = 'Base'
                """), {"tid": test_id}
        ).fetchall()

        # 3. Fetch Sub DeepDive Questions
        subs = db.session.execute(
            text("""
                    SELECT c.Category_Name, s.Sub_Text, a.Answer_Value
                    FROM dbo.Student_Test_Answer a
                    JOIN dbo.DeepDive_Sub_Questions s ON s.Sub_ID = a.Ref_ID
                    JOIN dbo.DeepDive_Base_Questions b ON b.Base_ID = s.Base_ID
                    JOIN dbo.Categories c ON c.Category_ID = b.Category_ID
                    WHERE a.Test_ID = :tid AND a.Item_Type = 'Sub'
                """), {"tid": test_id}
        ).fetchall()

        qa_list = []

        # Combine everything into one formatted list for the frontend
        for cat_name, q_text, val in gates + bases + subs:
            score = int(val)
            qa_list.append({
                "category": cat_name,
                "question": q_text,
                "score": score,
                "answer": likert_map.get(score, str(score))
            })

        return jsonify({"ok": True, "qa": qa_list})

    except Exception as e:
        print(f"Details error: {e}")
        return jsonify({"ok": False, "message": str(e)}), 500

@app.post("/api/daily-checkin")
def daily_checkin():
    data = request.json
    arid_no = data.get("arid_no")
    if not arid_no:
        return jsonify({"ok": False, "message": "Missing arid_no"}), 400

    try:
        sid = get_student_id_by_arid(arid_no)
        if not sid:
            return jsonify({"ok": False, "message": "Student not found"}), 404

        checkin_date = data.get("date")

        # 1. Save the Daily Metrics
        db.session.execute(
            text("""
                    MERGE dbo.Daily_CheckIn AS target
                    USING (VALUES (:sid, :date, :mood, :stress, :anxiety, :sleep, :energy, :notes, :stress_pct, :anxiety_pct, :depression_pct, :severity))
                    AS source (Student_ID, CheckIn_Date, Mood_Score, Stress_Score, Anxiety_Score, Sleep_Score, Energy_Score, Notes, Calculated_Stress_Pct, Calculated_Anxiety_Pct, Calculated_Depression_Pct, Severity_Label)
                    ON target.Student_ID = source.Student_ID AND target.CheckIn_Date = source.CheckIn_Date
                    WHEN MATCHED THEN 
                        UPDATE SET Mood_Score = source.Mood_Score, Stress_Score = source.Stress_Score, Anxiety_Score = source.Anxiety_Score,
                                   Sleep_Score = source.Sleep_Score, Energy_Score = source.Energy_Score, Notes = source.Notes,
                                   Calculated_Stress_Pct = source.Calculated_Stress_Pct, Calculated_Anxiety_Pct = source.Calculated_Anxiety_Pct,
                                   Calculated_Depression_Pct = source.Calculated_Depression_Pct, Severity_Label = source.Severity_Label
                    WHEN NOT MATCHED THEN
                        INSERT (Student_ID, CheckIn_Date, Mood_Score, Stress_Score, Anxiety_Score, Sleep_Score, Energy_Score, Notes, Calculated_Stress_Pct, Calculated_Anxiety_Pct, Calculated_Depression_Pct, Severity_Label)
                        VALUES (source.Student_ID, source.CheckIn_Date, source.Mood_Score, source.Stress_Score, source.Anxiety_Score, source.Sleep_Score, source.Energy_Score, source.Notes, source.Calculated_Stress_Pct, source.Calculated_Anxiety_Pct, source.Calculated_Depression_Pct, source.Severity_Label);
                """),
            {
                "sid": sid, "date": checkin_date,
                "mood": data.get("responses", {}).get("mood"), "stress": data.get("responses", {}).get("stress"),
                "anxiety": data.get("responses", {}).get("anxiety"), "sleep": data.get("responses", {}).get("sleep"),
                "energy": data.get("responses", {}).get("energy"), "notes": data.get("notes", ""),
                "stress_pct": data.get("stress"), "anxiety_pct": data.get("anxiety"),
                "depression_pct": data.get("depression"), "severity": data.get("severity")
            }
        )

        # 2. Save the Daily MLE Answers
        mle_answers = data.get("mle_answers", [])
        for ans in mle_answers:
            db.session.execute(
                text("""
                    INSERT INTO dbo.MLE_Answers (Student_ID, CheckIn_Date, Question_ID, Answer_Value)
                    VALUES (:sid, :date, :q_id, :val)
                """),
                {"sid": sid, "date": checkin_date, "q_id": ans["id"], "val": ans["value"]}
            )

        # 3. Update Completed Remedies
        completed_ids = data.get("completed_remedy_ids", [])
        if completed_ids:
            latest_test = db.session.execute(
                text("SELECT TOP 1 Test_ID FROM dbo.Test_Session WHERE Student_ID = :sid ORDER BY Started_On DESC"),
                {"sid": sid}).scalar()
            for rid in completed_ids:
                try:
                    db.session.execute(text(
                        "EXEC dbo.sp_RecordRemedyFeedback @Student_ID=:sid, @Test_ID=:tid, @Remedy_ID=:rid, @Status='Completed', @Rating=NULL, @Feedback_Text='Marked via Daily Check-in', @Dismissed_Reason=NULL"),
                                       {"sid": sid, "tid": latest_test, "rid": int(rid)})
                except ValueError:
                    pass

        db.session.commit()
        return jsonify({"ok": True, "message": "Daily check-in and MLEs saved!"})

    except Exception as e:
        db.session.rollback()
        print(f"Daily check-in error: {e}")
        return jsonify({"ok": False, "message": str(e)}), 500

@app.get("/api/student/suggested-remedies")
def get_student_suggested_remedies():
    """
    Returns the remedies that WOULD be suggested to a student based on their
    latest completed test session. Used by the advisor/psychologist profile modal
    to show what remedies are currently recommended for the student.
    """
    arid_no = request.args.get("arid_no")
    if not arid_no:
        return jsonify({"ok": False, "message": "Missing arid_no"}), 400

    try:
        student_id = get_student_id_by_arid(arid_no)
        if not student_id:
            return jsonify({"ok": False, "message": "Student not found"}), 404

        # Get the latest test session that has actual answers
        latest_test = db.session.execute(
            text("""
                SELECT TOP 1 ts.Test_ID
                FROM dbo.Test_Session ts
                JOIN dbo.Student_Test_Answer sta ON sta.Test_ID = ts.Test_ID
                WHERE ts.Student_ID = :sid
                ORDER BY ts.Started_On DESC
            """),
            {"sid": student_id}
        ).fetchone()

        if not latest_test:
            return jsonify({
                "ok": True,
                "remedies": [],
                "summary": None,
                "message": "No completed assessment found."
            })

        test_id = int(latest_test[0])

        # Build summary using existing scoring engine
        summary = build_test_summary(test_id)
        student_context = map_profile_context(student_id)

        # Fetch remedies using existing remedy engine (no duration cap for advisor view)
        remedies = safe_fetch_remedies(summary, student_context, max_duration=999)

        return jsonify({
            "ok": True,
            "remedies": remedies,
            "summary": {
                "dominant_disease": summary["Dominant_Disease"],
                "severity": summary["Severity"],
                "test_id": test_id,
                "bucket_pct": summary["bucket_pct"]
            }
        })

    except Exception as e:
        print(f"Student suggested remedies error: {e}")
        return jsonify({"ok": False, "message": str(e)}), 500
